if(MSVC)
	set_source_files_properties(bg_lib.cpp PROPERTIES COMPILE_OPTIONS /Oi-)

file(GLOB WOLF_GAME_SCAN
	"${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/ai/*.cpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/ai/fsm/*.cpp"
)
list(REMOVE_ITEM WOLF_GAME_SCAN "${CMAKE_CURRENT_SOURCE_DIR}/bg_lib.cpp")

add_custom_target(extractFunctionTables
	COMMAND $<TARGET_FILE:extractfuncs> ${WOLF_GAME_SCAN}
	WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
	DEPENDS extractfuncs
	COMMENT "Building the game function table"
	VERBATIM
)

add_library(${PROJECT_NAME} SHARED ${GAME_SOURCES})
set_target_properties(${PROJECT_NAME} PROPERTIES OUTPUT_NAME "qagame_${CPU_ARCH}")
target_compile_definitions(${PROJECT_NAME} PRIVATE GAMEDLL $<$<CONFIG:Debug>:BUILDING_REF_GL>)
add_dependencies(${PROJECT_NAME} extractFunctionTables)
if(WIN32)
	target_link_libraries(${PROJECT_NAME} PRIVATE winmm)
	target_link_options(${PROJECT_NAME} PRIVATE /BASE:0x20000000 /MAP:${ENGINE_MODULE_OUTPUT_DIRECTORY}/qagame_${CPU_ARCH}.map)
endif()
configure_target(${PROJECT_NAME})
